function onFaceResults(results) { if (results.multiFaceLandmarks && results.multiFaceLandmarks.length > 0) { // अगर MediaPipe का FaceMesh Matrix उपलब्ध है if (results.allMeshFaces && results.allMeshFaces[0]) { // डायरेक्ट मैट्रिक्स पास करने से 360-डिग्री रोटेशन और पोजीशन दोनों मिलती हैं } const landmarks = results.multiFaceLandmarks[0]; const nose = landmarks[1]; // नाक की टिप // पोजीशन सेट करें const x = -(nose.x - 0.5) * 4.0; const y = -(nose.y - 0.5) * 4.0; const z = -nose.z * 2.0; if (headModel) { headModel.visible = true; headModel.position.set(x, y, z); // सिर के चारों तरफ घूमने (Yaw, Pitch, Roll) के लिए सही लैंडमार्क पॉइंट्स से एंगल कैलकुलेट करें const leftEar = landmarks[234]; const rightEar = landmarks[454]; const upHead = landmarks[10]; const chin = landmarks[152]; // यहाँ हम इसे डायनेमिक रोटेशन दे रहे हैं ताकि सिर घुमाने पर मास्क भी घूमे const angleX = Math.atan2(chin.y - upHead.y, chin.z - upHead.z); const angleY = Math.atan2(rightEar.z - leftEar.z, rightEar.x - leftEar.x); const angleZ = Math.atan2(rightEar.y - leftEar.y, rightEar.x - leftEar.x); headModel.rotation.x = angleX; headModel.rotation.y = angleY; headModel.rotation.z = angleZ; } } else { if (headModel) { headModel.visible = false; } } renderer.render(scene, camera); }
