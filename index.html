// कैमरा चालू करने का फंक्शन async function startCamera() { try { if (mediaStream) { mediaStream.getTracks().forEach(track => track.stop()); } mediaStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: currentFacingMode, width: { ideal: 1280 }, height: { ideal: 720 } }, audio: true }); video.srcObject = mediaStream; message("कैमरा सफलतापूर्वक शुरू हो गया है।"); $("start").disabled = true; $("switchCam").disabled = false; $("record").disabled = false; $("stop").disabled = false; } catch (err) { console.error(err); message("कैमरा शुरू करने में त्रुटि: " + err.message); } } // कैमरा बंद करने का फंक्शन function stopCamera() { if (mediaStream) { mediaStream.getTracks().forEach(track => track.stop()); mediaStream = null; } video.srcObject = null; message("कैमरा बंद कर दिया गया है।"); $("start").disabled = false; $("switchCam").disabled = true; $("record").disabled = true; $("stop").disabled = true; } $("start").addEventListener("click", startCamera); $("stop").addEventListener("click", stopCamera); $("switchCam").addEventListener("click", () => { currentFacingMode = currentFacingMode === "user" ? "environment" : "user"; startCamera(); }); // पेज लोड होते ही बटन इनेबल करना window.addEventListener("DOMContentLoaded", () => { $("start").disabled = false; message("लाइब्रेरी लोड हो गई हैं। कृपया 'Camera Start' पर क्लिक करें।"); }); // एनीमेशन लूप function animate() { requestAnimationFrame(animate); renderer.render(scene, camera); } animate();
